### Unit do monitor
$ systemctl cat citizentech-monitor.service
# /etc/systemd/system/citizentech-monitor.service
# =============================================================================
# CitizenTech - Serviço de monitoramento (CPU, memória RAM, disco e aplicação)
# Template: os campos delimitados por '@' são preenchidos por 03-provision-monitoring.sh
#
# Tolerância a falhas em 3 camadas:
#   1. Restart=always      -> se o processo morrer (crash, kill -9), o systemd
#                             reinicia em 5s
#   2. StartLimitBurst     -> se entrar em loop de falhas (5 em 2 min), o
#                             systemd para de tentar e marca 'failed'
#   3. Watchdog (timer)    -> citizentech-healthcheck.timer detecta o serviço
#                             parado/failed/travado e o recupera
# =============================================================================
[Unit]
Description=CitizenTech - Monitoramento de CPU, memoria e disco
After=network.target citizentech-network.service
StartLimitIntervalSec=120
StartLimitBurst=5

[Service]
Type=simple
ExecStartPre=-+/opt/citizentech/bin/service-event.sh start %n
ExecStart=/opt/citizentech/bin/monitor.sh --daemon
ExecStopPost=-+/opt/citizentech/bin/service-event.sh stop %n
Restart=always
RestartSec=5
TimeoutStopSec=15
# Prioridade reduzida: o monitor não deve competir com a aplicação
Nice=10
SyslogIdentifier=citizentech-monitor

# Endurecimento de segurança
NoNewPrivileges=yes
PrivateTmp=yes
ProtectSystem=strict
ProtectHome=read-only
ProtectKernelModules=yes
ReadWritePaths=/var/log/citizentech /var/lib/citizentech/state

[Install]
WantedBy=multi-user.target

### Unit da aplicação
$ systemctl cat citizentech-app.service
# /etc/systemd/system/citizentech-app.service
# =============================================================================
# CitizenTech - Aplicação Web
# Template: os campos delimitados por '@' são preenchidos por 02-provision-app.sh
# =============================================================================
[Unit]
Description=CitizenTech - Aplicacao Web (citizentech-app, runtime python)
Requires=citizentech-network.service
After=network.target citizentech-network.service
# Tolerância a falhas: até 10 reinícios em 2 minutos antes de desistir
# (depois disso o watchdog citizentech-healthcheck.timer assume a recuperação)
StartLimitIntervalSec=120
StartLimitBurst=10

[Service]
Type=simple
User=citizentech
Group=citizentech
WorkingDirectory=/opt/citizentech/app
EnvironmentFile=/etc/citizentech/app.env
# '-+' = executa como root e nunca impede o serviço de subir
ExecStartPre=-+/opt/citizentech/bin/service-event.sh start %n
ExecStart=/opt/citizentech/venv/bin/gunicorn --workers 2 --bind 10.77.0.10:8000 --timeout 30 --access-logfile /var/log/citizentech/app/access.log --error-logfile - --capture-output app:app
ExecStopPost=-+/opt/citizentech/bin/service-event.sh stop %n
Restart=always
RestartSec=3
TimeoutStopSec=20
UMask=0027
SyslogIdentifier=citizentech-app

# Endurecimento de segurança (sandbox do systemd)
NoNewPrivileges=yes
PrivateTmp=yes
ProtectSystem=strict
ProtectHome=yes
ProtectKernelTunables=yes
ProtectKernelModules=yes
ProtectControlGroups=yes
ReadWritePaths=/var/lib/citizentech/db /var/lib/citizentech/home /var/log/citizentech/app

[Install]
WantedBy=multi-user.target
